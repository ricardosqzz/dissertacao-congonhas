  144  def link_receptors(coords,zones,grid):
  145      """Recupera posição original somente quando a exportação arredondada coincide.
  146  
  147      A concentração permanece vinculada ao X/Y do POST. Não usa vizinho mais
  148      próximo nem amplia zonas; guarda evidência da correção subcentimétrica.
  149      """
  150      import geopandas as gpd
  151      original=grid.to_crs(zones.crs).copy()
  152      original['X']=original.geometry.x.round(2);original['Y']=original.geometry.y.round(2)
  153      if original.duplicated(['X','Y']).any():raise ValueError('Grade ambígua após arredondamento')
  154      xy=coords[['X','Y']].drop_duplicates().copy()
  155      p=gpd.GeoDataFrame(xy,geometry=gpd.points_from_xy(xy.X,xy.Y),crs=zones.crs)
  156      initial=gpd.sjoin(p,zones[['zone_index','geometry']],how='inner',predicate='within')
  157      merged=xy.merge(original[['X','Y','geometry']],how='left',on=['X','Y'])
  158      resolved=merged.geometry.notna();distance=gpd.GeoSeries(merged.loc[resolved,'geometry'],crs=zones.crs).distance(p.loc[resolved,'geometry'])
  159      if distance.max()>.0072:raise ValueError('Deslocamento incompatível com arredondamento centimétrico')
  160      p.loc[resolved,'geometry']=merged.loc[resolved,'geometry'].to_numpy()
  161      strict=gpd.sjoin(p,zones[['zone_index','geometry']],how='inner',predicate='within')[['X','Y','zone_index']]
  162  
  163      boundary=zones[['zone_index','geometry']].copy();boundary.geometry=boundary.geometry.buffer(1e-7)
  164      links=gpd.sjoin(p,boundary,how='inner',predicate='intersects')[['X','Y','zone_index']]
  165      keys=set(map(tuple,strict[['X','Y','zone_index']].to_numpy()))
  166      links['spatial_support']=['interior' if tuple(r) in keys else 'fronteira_numerica' for r in links[['X','Y','zone_index']].to_numpy()]
