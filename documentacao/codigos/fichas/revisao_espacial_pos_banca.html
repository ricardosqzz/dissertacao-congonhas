    Cálculos a observar nesta rotina.

  • mean: Média no eixo explicitado. Não representa automaticamente média anual ou diária sem a seleção temporal correspondente.
  • sum: Soma no eixo definido. Em suportes sobrepostos, uma soma entre vizinhanças pode contar o mesmo local repetidas vezes.
  • std: Desvio-padrão. Na padronização z, coloca colunas em escala comparável; conferir ddof no trecho.
  • cdist: Matriz de distâncias entre observações. É usada no agrupamento, na silhouette ou na comparação entre centroides.
  • quantile: Quantil da amostra fornecida. O eixo e o conjunto de dados da chamada definem se o quantil é temporal ou espacial.
  • log1p: Calcula ln(1+x), reduzindo assimetria e preservando zero. O resultado transformado não mantém a unidade física original.
  • sqrt: Raiz quadrada. No ajuste ponderado do k-means, multiplicar a coluna por √peso faz o peso entrar uma única vez na distância ao quadrado.
  • average: Média com ou sem pesos. As unidades do resultado são as da variável antes de eventual normalização.
  • linkage: Agrupamento hierárquico. Com method=ward, as fusões procuram o menor aumento da dispersão interna.
  • fcluster: Transforma a árvore hierárquica em rótulos de grupos segundo o critério fornecido.
  • kmeans2: Ajusta centroides pelo k-means. A matriz de entrada, a inicialização ++, a semente e as repetições determinam a solução reproduzida.

  106  def local_moran(x,ids,y=None,permutations=999,seed=SEED):

  157  # Emission allocation
  158  def emission_allocation(cells,municipality,pol):
  159   sources=gpd.read_file(BASE/f'output/13_rateio_emissoes/fontes_rateio_{pol}.gpkg').to_crs(cells.crs)
  160   masses=np.zeros(len(cells));records=[];expected=0.;missing=0
  161   for r in sources.itertuples():
  162    geom=make_valid(r.geometry)
  163    if geom.is_empty or not np.isfinite(r.massa):missing+=1;continue
  164    mode='area' if geom.area>0 else 'length' if geom.length>0 else 'point'
  165    denominator=geom.area if mode=='area' else geom.length if mode=='length' else 1
  166    inside=geom.intersection(municipality)
  167    fraction=inside.area/denominator if mode=='area' else inside.length/denominator if mode=='length' else float(municipality.covers(geom))
  168    expected+=r.massa*fraction
  169    candidates=cells.sindex.query(geom,predicate='intersects')
  170    if mode=='point' and len(candidates)>1:
  171     candidates=np.array([min(candidates,key=lambda j:(cells.geometry.iloc[j].representative_point().distance(geom),j))])
  172    for j in candidates:
  173     it=geom.intersection(cells.geometry.iloc[j]);f=it.area/denominator if mode=='area' else it.length/denominator if mode=='length' else 1.
  174     if f<=0:continue
  175     mass=float(r.massa*f);masses[j]+=mass;records.append({'pollutant':pol,'source_id':r.source_id,'group_id':r.group_id,'source_type':r.source_type,'receptor_id':int(cells.receptor_id.iloc[j]),'fraction':f,'mass_tpy':mass})
  176   err=abs(masses.sum()-expected);assert err<=max(.001,expected*1e-7),(pol,err,expected)
  177   pd.DataFrame(records).to_csv(OUT/f'{pol}_fontes_celulas.csv',index=False)
  178   return masses,{'total_source_mass_tpy':float(sources.massa.sum()),'municipal_mass_expected_tpy':expected,'allocated_mass_tpy':float(masses.sum()),'absolute_error_tpy':err,'sources_missing_geometry_or_mass':missing,'volume_geometry':'Suporte geométrico do cadastro; não área física emissora comprovada.'}

  180  # Stump
  181  def stump(x,y,min_fraction=.1):
  182   order=np.argsort(x);sx=x[order];sy=y[order];n=len(x);a=np.cumsum(sy);b=np.cumsum(sy*sy);j=np.arange(1,n);minn=max(5,int(n*min_fraction))
  183   score=(b[:-1]-a[:-1]**2/j)+(b[-1]-b[:-1]-(a[-1]-a[:-1])**2/(n-j));ok=(j>=minn)&(j<=n-minn)&(sx[:-1]<sx[1:]);score[~ok]=np.inf
  184   if not np.isfinite(score).any():return None
  185   cut=np.argmin(score);return (float((sx[cut]+sx[cut+1])/2),float(sy[:cut+1].mean()),float(sy[cut+1:].mean()))

  187  # Thresholds
  188  def thresholds(x,y,xy,pol,unit):
  189   out=[];lx=np.log1p(x);ly=np.log1p(y);fit=stump(lx,ly)
  190   for size in [2000,4000]:
  191    blocks=np.floor((xy-xy.min(0))/size).astype(int);keys=[tuple(v) for v in blocks];unique=sorted(set(keys));rng=np.random.default_rng(SEED);rng.shuffle(unique);fold={b:i%5 for i,b in enumerate(unique)};folds=np.array([fold[k] for k in keys])
  192    for k in range(5):
  193     test=folds==k;train=~test
  194     if min(test.sum(),train.sum())<10:continue
  195     f=stump(lx[train],ly[train]);
  196     if f is None:continue
  197     pred=np.where(lx[test]<=f[0],f[1],f[2]);mse=np.mean((pred-ly[test])**2);null=np.mean((ly[train].mean()-ly[test])**2)
  198     out.append({'pollutant':pol,'emission_basis':unit,'block_m':size,'fold':k,'threshold':np.expm1(f[0]),'threshold_all_data':np.expm1(fit[0]) if fit else np.nan,'skill_vs_mean':1-mse/null if null>0 else np.nan,'rmse_log1p':np.sqrt(mse),'n_train':int(train.sum()),'n_test':int(test.sum()),'low_mean_conc':np.expm1(f[1]),'high_mean_conc':np.expm1(f[2])})
  199   return out
