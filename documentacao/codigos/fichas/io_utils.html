   53  def _read_kml_with_encoding_fallback(path: Path):
   54      try:
   55          return gpd.read_file(path)
   56      except Exception as first_error:
   57          raw = path.read_bytes()
   58          last_error = first_error

   61                  text = raw.decode(encoding)

   66                      return gpd.read_file(tmp_path)
   67                  finally:
   68                      tmp_path.unlink(missing_ok=True)
   69              except Exception as exc:
   70                  last_error = exc

   85                  zf.extractall(temp)

  109  def normalize_name(s):
  110      s = str(s or "").strip().lower()
  111      s = s.replace("ç", "c").replace("ã", "a").replace("á", "a").replace("à", "a")
  112      s = s.replace("é", "e").replace("ê", "e").replace("í", "i").replace("ó", "o")
  113      s = s.replace("ô", "o").replace("õ", "o").replace("ú", "u")
  114      return re.sub(r"[^a-z0-9]+", "-", s).strip("-")

  116  # Excel serial to datetime
  117  def excel_serial_to_datetime(series: pd.Series):
  118      """Converte datas robustamente, inclusive serial Excel."""
  119      s = series.copy()
  120      numeric = pd.to_numeric(s, errors="coerce")
  121      numeric_share = numeric.notna().mean() if len(s) else 0
  122      if numeric_share > 0.8 and numeric.dropna().between(20000, 80000).mean() > 0.8:
  123          return pd.to_datetime(numeric, unit="D", origin="1899-12-30", errors="coerce")
  124      return pd.to_datetime(s, dayfirst=True, errors="coerce")

  126  # Sanitize vector columns
  127  def _sanitize_vector_columns(gdf):
  128      out = gdf.copy()
  129      rename = {}
  130      drop_cols = []
  131      for c in out.columns:
  132          if c == out.geometry.name:
  133              continue
  134          lc = str(c).strip().lower()
  135          if lc in {"fid", "ogc_fid"}:
  136              rename[c] = f"src_{lc}"
  137          elif lc in {"geometry"}:
  138              drop_cols.append(c)
  139      if drop_cols:
  140          out = out.drop(columns=drop_cols, errors="ignore")
  141      if rename:
  142          out = out.rename(columns=rename)
  143      return out

  151  # Shp safe columns
  152  def _shp_safe_columns(gdf):
  153      """Renomeia atributos para <=10 caracteres e evita colisões do Shapefile."""
  154      out = gdf.copy()
  155      geom = out.geometry.name
  156      used = set()
  157      mapping = {}
  158      rename = {}
  159      for c in out.columns:
  160          if c == geom:
  161              continue
  162          raw = str(c)
  163          base = re.sub(r"[^A-Za-z0-9_]", "_", raw).upper().strip("_") or "FIELD"
  164          candidate = base[:10]
  165          if candidate in used:
  166              prefix = base[:7]
  167              i = 1
  168              while True:
  169                  candidate = f"{prefix}{i:03d}"[:10]
  170                  if candidate not in used:
  171                      break
  172                  i += 1
  173          used.add(candidate)
  174          rename[c] = candidate
  175          mapping[raw] = candidate
  176      return out.rename(columns=rename), mapping
