   50          dates=pd.date_range(pd.to_datetime(str(first_code//100),format='%y%m%d')+pd.Timedelta(hours=first_code%100-1),
   51                              pd.to_datetime(str(last_code//100),format='%y%m%d')+pd.Timedelta(hours=last_code%100-1),
   52                              freq='h' if self.info['period']=='1H' else 'D')
   53          proposed=(((dates.year%100*100+dates.month)*100+dates.day)*100+dates.hour+1).to_numpy(dtype=np.int64)
