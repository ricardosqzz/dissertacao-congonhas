   56  # Nearest rank percentile
   57  def nearest_rank_percentile(values, p=0.98):
   58      """Percentil por posto (nearest-rank), sem interpolação.
   59  
   60      Para n valores ordenados crescentemente, usa posição ceil(p*n), limitada a 1..n.
   61      Essa forma é mais próxima do procedimento de seleção por posto usado em métricas
   62      regulatórias do que a interpolação linear padrão de muitas bibliotecas.
   63      """
   64      import math
   65      import numpy as np
   66      arr = np.asarray(list(values), dtype=float)
   67      arr = arr[np.isfinite(arr)]
   68      if arr.size == 0:
   69          return np.nan
   70      arr.sort()
   71      rank = max(1, min(arr.size, int(math.ceil(float(p) * arr.size))))
   72      return float(arr[rank - 1])
