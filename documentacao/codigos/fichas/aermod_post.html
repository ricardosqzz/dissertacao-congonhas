   64      out = []

  113  def iter_pos_chunks(path: Path, chunk_rows=350_000, progress=False, progress_label=None):
  114      """Itera pelas linhas de dados do POST em blocos.
  115  
  116      Quando ``progress=True``, imprime progresso aproximado por tamanho lido,
  117      número de registros válidos e tempo decorrido. O objetivo é deixar claro
  118      que arquivos .POS gigantes continuam sendo processados.
  119      """
  120      path = Path(path)
  121      rows = []
  122      data_rows = 0
  123      raw_lines = 0
  124      bytes_seen = 0
  125      file_size = path.stat().st_size if path.exists() else 0
  126      t0 = time.perf_counter()
  127      last_report = t0
  128      min_seconds = float(getattr(config, "POS_PROGRESS_MIN_SECONDS", 8.0))
  129      label = progress_label or path.name
  130  
  131      with path.open("r", encoding="latin1", errors="replace", newline="") as f:
  132          for line in f:
  133              raw_lines += 1
  134  
  135              bytes_seen += len(line)
  136              if not DATA_RE.match(line):
  137                  continue
  138              p = line.split()
  139  
  140              if len(p) < 9:
  141                  continue
  142              try:
  143                  rows.append((
  144                      float(p[0]), float(p[1]), float(p[2]),
  145                      p[6], p[7], str(p[8]).zfill(8)
  146                  ))
  147                  data_rows += 1
  148              except Exception:
  149                  continue
  150              if len(rows) >= chunk_rows:
  151                  now = time.perf_counter()
  152                  if progress and (now - last_report >= min_seconds):
  153                      pct = min(100.0, 100.0 * bytes_seen / file_size) if file_size else 0.0
  154                      print(
  155                          f"      [PROGRESSO] {label}: {pct:5.1f}% | "
  156                          f"{data_rows:,} registros | {_fmt_size(bytes_seen)} de {_fmt_size(file_size)} | "
  157                          f"tempo {_fmt_elapsed(now-t0)}",
  158                          flush=True,
  159                      )
  160                      last_report = now
  161                  yield pd.DataFrame(rows, columns=["X", "Y", "CONC", "AVE", "GRP", "DATECODE"])
  162                  rows = []
  163      if rows:
  164          yield pd.DataFrame(rows, columns=["X", "Y", "CONC", "AVE", "GRP", "DATECODE"])
  165      if progress:
  166          elapsed = time.perf_counter() - t0
  167          print(
  168              f"      [PROGRESSO] {label}: 100.0% | {data_rows:,} registros válidos lidos | "
  169              f"{_fmt_size(file_size)} | tempo {_fmt_elapsed(elapsed)}",
  170              flush=True,
  171          )

  384  def read_post_hourly_for_day(path: Path, target_date, background=0.0, chunk_rows=350_000, progress=False):
  385      """Extrai a série horária de um único dia a partir de POST 1-HR."""
  386      target = pd.Timestamp(target_date).normalize()
  387      parts = []
  388      for ch in iter_pos_chunks(path, chunk_rows=chunk_rows, progress=progress, progress_label=path.name):
  389          if ch.empty:
  390              continue
  391          ch["_kind"] = ch["AVE"].map(_ave_kind)
  392          ch = ch[ch["_kind"] == "1H"].copy()
  393          if ch.empty:
  394              continue
  395          dt, hh = parse_datecode(ch["DATECODE"])
  396          ch["date"] = dt
  397          ch["hour"] = pd.to_numeric(hh, errors="coerce")
  398          ch["CONC"] = pd.to_numeric(ch["CONC"], errors="coerce") + float(background)
  399          ch = ch.dropna(subset=["X", "Y", "CONC", "date", "hour"])
  400          ch = ch[ch["date"] == target]
  401          if ch.empty:
  402              continue
  403          ch = ch[["X", "Y", "CONC", "date", "hour", "GRP"]].copy()
  404          parts.append(ch)
  405      if not parts:
  406          return pd.DataFrame(columns=["X","Y","CONC","date","hour","GRP"])
  407      out = pd.concat(parts, ignore_index=True)
  408      out = out.groupby(["X","Y","date","hour","GRP"], observed=True, as_index=False)["CONC"].mean()
  409      return out

  414      if t in {"24", "24HR", "24H"} or t.startswith("24HR"):

  420  # Read post daily
  421  def read_post_daily(path: Path, background=0.0, chunk_rows=350_000, progress=False, progress_label=None):
  422      """Retorna uma série diária por receptor a partir de POST 24-HR ou 1-HR.
  423  
  424      Colunas: X, Y, CONC, date, year, GRP, N_HOURS, SOURCE_AVERAGING.
  425      Para POST 1-HR, um dia é mantido apenas quando possui pelo menos
  426      ``config.MIN_VALID_HOURS_DAY`` horas. Para POST 24-HR, o valor é usado
  427      diretamente como média de 24 h.
  428      """
  429      info = inspect_pos_header(path)
  430      direct_parts = []
  431      hourly_parts = []
  432      detected = set()
  433  
  434      for ch in iter_pos_chunks(path, chunk_rows=chunk_rows, progress=progress, progress_label=progress_label):
  435          if ch.empty:
  436              continue
  437          ch["_kind"] = ch["AVE"].map(_ave_kind)
  438          detected.update(ch["_kind"].dropna().unique().tolist())
  439          ch = ch[ch["_kind"].isin(["1H", "24H"])].copy()
  440          if ch.empty:
  441              continue
  442          code = ch["DATECODE"].astype(str).str.replace(r"\.0$", "", regex=True).str.zfill(8)
  443          yy = pd.to_numeric(code.str.slice(0, 2), errors="coerce")
  444          mm = pd.to_numeric(code.str.slice(2, 4), errors="coerce")
  445          dd = pd.to_numeric(code.str.slice(4, 6), errors="coerce")
  446          ch["date"] = pd.to_datetime(dict(year=2000 + yy, month=mm, day=dd), errors="coerce")
  447          ch["CONC"] = pd.to_numeric(ch["CONC"], errors="coerce") + float(background)
  448          ch = ch.dropna(subset=["X", "Y", "CONC", "date"])
  449  
  450          d24 = ch[ch["_kind"] == "24H"]
  451          if not d24.empty:
  452  
  453              a = d24.groupby(["X", "Y", "date", "GRP"], observed=True, as_index=False).agg(CONC=("CONC", "mean"))
  454              a["N_HOURS"] = 24
  455              direct_parts.append(a)
  456  
  457          d1 = ch[ch["_kind"] == "1H"]
  458          if not d1.empty:
  459  
  460              a = d1.groupby(["X", "Y", "date", "GRP"], observed=True, as_index=False).agg(
  461                  SUM_CONC=("CONC", "sum"), N_HOURS=("CONC", "count")
  462              )
  463              hourly_parts.append(a)
  464  
  465      if direct_parts:
  466          daily = pd.concat(direct_parts, ignore_index=True)
  467          daily = daily.groupby(["X", "Y", "date", "GRP"], observed=True, as_index=False).agg(
  468              CONC=("CONC", "mean"), N_HOURS=("N_HOURS", "max")
  469          )
  470          source_ave = "24H"
  471      elif hourly_parts:
  472          h = pd.concat(hourly_parts, ignore_index=True)
  473          h = h.groupby(["X", "Y", "date", "GRP"], observed=True, as_index=False).agg(
  474              SUM_CONC=("SUM_CONC", "sum"), N_HOURS=("N_HOURS", "sum")
  475          )
  476          h = h[h["N_HOURS"] >= int(config.MIN_VALID_HOURS_DAY)].copy()
  477          h["CONC"] = h["SUM_CONC"] / h["N_HOURS"]
  478          daily = h[["X", "Y", "date", "GRP", "CONC", "N_HOURS"]]
  479          source_ave = "1H→24H"
  480      else:
  481          raise ValueError(f"Nenhum registro 1-HR ou 24-HR encontrado em {path}")
  482  
  483      daily["year"] = daily["date"].dt.year.astype("Int64")
  484      daily["SOURCE_AVERAGING"] = source_ave
  485      info = dict(info)
  486      info["daily_source"] = source_ave
  487      info["detected_averaging"] = ",".join(sorted(detected))
  488      return daily, info

  497  # Pos fingerprint
  498  def _pos_fingerprint(path: Path):
  499      path = Path(path)
  500      st = path.stat()
  501      n = int(getattr(config, "POS_CACHE_SAMPLE_BYTES", 1024 * 1024))
  502      h = hashlib.sha256()
  503      with path.open("rb") as f:
  504          head = f.read(n)
  505          h.update(head)
  506          if st.st_size > n:
  507              f.seek(max(0, st.st_size - n))
  508              h.update(f.read(n))
  509      return {
  510          "size": int(st.st_size),
  511          "mtime_ns": int(st.st_mtime_ns),
  512          "sample_sha256": h.hexdigest(),
  513      }

  515  # Pos cache paths
  516  def _pos_cache_paths(path: Path, pollutant=None):
  517      cache_root = Path(getattr(config, "DIRS", {}).get("pos_cache", Path(config.BASE) / "cache" / "pos"))
  518      pol = _safe_pollutant_name(pollutant, path)
  519      folder = cache_root / pol
  520      folder.mkdir(parents=True, exist_ok=True)
  521      resolved = Path(path).resolve()
  522      try:
  523          identity = resolved.relative_to(Path(config.BASE).resolve()).as_posix().lower()
  524      except ValueError:
  525          identity = str(resolved).lower()
  526      key = hashlib.sha1(identity.encode("utf-8", errors="ignore")).hexdigest()[:10]
  527      stem = re.sub(r"[^A-Za-z0-9_.-]+", "_", Path(path).stem)
  528      base = folder / f"{stem}_{key}"
  529      return {
  530          "daily": Path(str(base) + "_daily.pkl.gz"),
  531          "meta": Path(str(base) + "_meta.json"),
  532      }

  542  # Cache is valid
  543  def _cache_is_valid(path: Path, meta_path: Path, background):
  544      if not meta_path.exists():
  545          return False, None
  546      try:
  547          meta = json.loads(meta_path.read_text(encoding="utf-8"))
  548      except Exception:
  549          return False, None
  550      if meta.get("signature") != _cache_signature(background):
  551          return False, meta
  552      try:
  553          fp = _pos_fingerprint(path)
  554      except Exception:
  555          return False, meta
  556      return meta.get("source_fingerprint") == fp, meta

  558  # Read post daily cached
  559  def read_post_daily_cached(path: Path, background=0.0, pollutant=None, chunk_rows=350_000):
  560      """Lê o POST e mantém cache persistente da série diária tratada.
  561  
  562      O cache é invalidado automaticamente quando o .POS muda em tamanho, data
  563      de modificação ou amostra binária do início/fim.
  564      """
  565      path = Path(path)
  566      use_cache = bool(getattr(config, "USE_POS_CACHE", True))
  567      force = bool(getattr(config, "FORCE_REBUILD_POS_CACHE", False))
  568      paths = _pos_cache_paths(path, pollutant)
  569  
  570      if use_cache and not force and paths["daily"].exists():
  571          valid, meta = _cache_is_valid(path, paths["meta"], background)
  572          if valid:
  573              t0 = time.perf_counter()
  574              print(f"    [CACHE HIT] {path.name} -> usando série diária tratada", flush=True)
  575              daily = pd.read_pickle(paths["daily"], compression="gzip")
  576              load_elapsed = time.perf_counter() - t0
  577              print(
  578                  f"    [CACHE OK] {path.name}: {len(daily):,} registros diários carregados em "
  579                  f"{_fmt_elapsed(load_elapsed)}", flush=True
  580              )
  581              info = dict((meta or {}).get("info") or {})
  582              info["cache_used"] = True
  583              info["cache_file"] = str(paths["daily"])
  584              info["cache_load_seconds"] = round(time.perf_counter() - t0, 3)
  585              return daily, info
  586  
  587      reason = "forçado" if force else "primeira leitura ou .POS alterado"
  588      print(f"    [CACHE MISS] {path.name} -> {reason}; processando arquivo bruto...", flush=True)
  589      t0 = time.perf_counter()
  590      progress_enabled = bool(getattr(config, "SHOW_POS_PROGRESS", True))
  591      chunk_rows = int(getattr(config, "POS_PROGRESS_CHUNK_ROWS", chunk_rows))
  592      daily, info = read_post_daily(
  593          path, background=background, chunk_rows=chunk_rows,
  594          progress=progress_enabled, progress_label=path.name,
  595      )
  596      elapsed = time.perf_counter() - t0
  597  
  598      if use_cache:
  599          t1 = time.perf_counter()
  600  
  601          tmp_daily = Path(str(paths["daily"]) + ".tmp")
  602          tmp_meta = Path(str(paths["meta"]) + ".tmp")
  603          for tmp in (tmp_daily, tmp_meta):
  604              if tmp.exists():
  605                  try:
  606                      tmp.unlink()
  607                  except OSError:
  608                      pass
  609          daily.to_pickle(tmp_daily, compression="gzip")
  610          meta = {
  611              "source_file": str(path.resolve()),
  612              "source_fingerprint": _pos_fingerprint(path),
  613              "signature": _cache_signature(background),
  614              "info": dict(info),
  615              "rows_daily": int(len(daily)),
  616              "created_epoch": time.time(),
  617              "raw_processing_seconds": round(elapsed, 3),
  618          }
  619          tmp_meta.write_text(json.dumps(meta, indent=2, ensure_ascii=False), encoding="utf-8")
  620          os.replace(tmp_daily, paths["daily"])
  621          os.replace(tmp_meta, paths["meta"])
  622          print(
  623              f"    [CACHE SAVE] {path.name}: {len(daily):,} registros diários tratados "
  624              f"salvos em {paths['daily'].name} ({time.perf_counter()-t1:.1f}s)",
  625              flush=True,
  626          )
  627      info = dict(info)
  628      info["cache_used"] = False
  629      info["cache_file"] = str(paths["daily"]) if use_cache else ""
  630      info["raw_processing_seconds"] = round(elapsed, 3)
  631      return daily, info

  633  # Summarize post for pm
  634  def summarize_post_for_pm(path: Path, pollutant: str, background=0.0):
  635      """Calcula P98 diário e média anual por receptor usando POST 24h ou horário."""
  636      daily, info = read_post_daily_cached(path, background=background, pollutant=pollutant)
  637      yearly_parts = []
  638      for year, dyear in daily.groupby("year", observed=True):
  639          agg = dyear.groupby(["X", "Y"], observed=True).agg(
  640              ANNUAL_MEAN=("CONC", "mean"),
  641              N_DAYS=("CONC", "count"),
  642              P98=("CONC", lambda x: nearest_rank_percentile(x, config.SHORT_TERM_PERCENTILE)),
  643          ).reset_index()
  644          year = int(year)
  645          expected = 366 if pd.Timestamp(year, 12, 31).is_leap_year else 365
  646          agg["year"] = year
  647          agg["EXPECTED_DAYS"] = expected
  648          agg["COMPLETENESS"] = agg["N_DAYS"] / expected
  649          agg["VALID_YEAR"] = agg["COMPLETENESS"] >= config.MIN_QUARTER_COMPLETENESS
  650          yearly_parts.append(agg)
  651  
  652      if not yearly_parts:
  653          raise ValueError(f"Sem médias diárias válidas em {path}")
  654      yearly = pd.concat(yearly_parts, ignore_index=True)
  655      n_required = int(config.MODEL_N_YEARS_DESIGN)
  656  
  657      summaries = []
  658      for (x, y), sub in yearly.groupby(["X", "Y"], sort=False):
  659          valid_years = sorted(sub.loc[sub["VALID_YEAR"], "year"].astype(int).tolist())
  660          valid = pd.DataFrame(columns=sub.columns)
  661          for end_year in sorted(valid_years, reverse=True):
  662              wanted = list(range(end_year - n_required + 1, end_year + 1))
  663              if all(yv in valid_years for yv in wanted):
  664                  valid = sub[sub["year"].isin(wanted)].sort_values("year")
  665                  break
  666          row = {
  667              "X": float(x), "Y": float(y),
  668              "n_model_years": int(len(valid)),
  669              "years_model": ",".join(map(str, valid["year"].astype(int).tolist())) if len(valid) else "",
  670              "SOURCE_AVERAGING": info.get("daily_source", ""),
  671          }
  672          if len(valid) == n_required:
  673              row["P98_MODEL"] = float(valid["P98"].mean())
  674              row["ANNUAL_MODEL"] = float(valid["ANNUAL_MEAN"].mean())
  675          else:
  676              row["P98_MODEL"] = np.nan
  677              row["ANNUAL_MODEL"] = np.nan
  678          row["n_design_years"] = row["n_model_years"]
  679          row["years_valid"] = row["years_model"]
  680          row["P98_3YR"] = row["P98_MODEL"]
  681          row["ANNUAL_3YR"] = row["ANNUAL_MODEL"]
  682          summaries.append(row)
  683  
  684      out = pd.DataFrame(summaries)
  685      for scenario in ("PV", "PF"):
  686          lim24 = STANDARDS[pollutant][scenario]["24H"]
  687          limann = STANDARDS[pollutant][scenario]["ANNUAL"]
  688          out[f"ratio24_{scenario}"] = out["P98_MODEL"] / lim24
  689          out[f"ratioann_{scenario}"] = out["ANNUAL_MODEL"] / limann
  690          out[f"status_{scenario}"] = np.select(
  691              [
  692                  out["n_model_years"] < n_required,
  693                  (out["P98_MODEL"] <= lim24) & (out["ANNUAL_MODEL"] <= limann),
  694              ],
  695              ["insuficiente", "atendimento"],
  696              default="nao_atendimento",
  697          )
  698  
  699      gdf = gpd.GeoDataFrame(out, geometry=gpd.points_from_xy(out["X"], out["Y"]), crs=config.CRS_PROJECT)
  700      return gdf, yearly, info

  712  def discover_post_main(post_dir: Path, pollutant: str):
  713      """Localiza o cenário ALL do poluente.
  714  
  715      Prioriza arquivos do grupo ALL dentro da pasta do poluente e aceita tanto
  716      POST 24-HR quanto POST 1-HR. O período é verificado pelo cabeçalho.
  717      """
  718      override = config.POST_OVERRIDE[pollutant].get("24H")
  719      if override:
  720          p = Path(override)
  721          if p.exists():
  722              return p
  723      folder = _pollutant_folder(post_dir, pollutant)
  724      files = _unique_paths(sorted(folder.glob("*.POS")) + sorted(folder.glob("*.pos")))
  725      if not files and folder == post_dir:
  726          files = _unique_paths(sorted(post_dir.rglob("*.POS")) + sorted(post_dir.rglob("*.pos")))
  727  
  728      compatible = []
  729      for p in files:
  730          pol = _infer_pollutant(p)
  731          if pol not in {None, pollutant}:
  732              continue
  733          info = inspect_pos_header(p)
  734          kind = _ave_kind(info.get("ave"))
  735          if kind not in {"1H", "24H"}:
  736              continue
  737          text = re.sub(r"[^A-Z0-9]+", "", p.stem.upper())
  738          is_all = "GALL" in text or text.endswith("ALL") or str(info.get("group", "")).upper() == "ALL"
  739          compatible.append((p, kind, is_all))
  740  
  741      all_files = [x for x in compatible if x[2]]
  742      pool = all_files or compatible
  743      if not pool:
  744          return None
  745  
  746      pool.sort(key=lambda x: (0 if x[1] == "24H" else 1, x[0].name))
  747      return pool[0][0]
