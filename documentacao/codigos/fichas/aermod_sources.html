codigo/aermod_sources.py

Reconstrói geometrias das fontes das planilhas/KML, interpreta o tipo de taxa, associa empreendimentos, aloca massa às zonas e confere conservação. Fluxos superficiais precisam da área emissora para resultar em g/s; fontes com taxa já integrada não recebem uma multiplicação adicional. O código distingue fontes de diferentes tipologias antes da distribuição territorial.

  299          if np.isfinite(area_m2) and area_m2 > 0:
  300              return rate * area_m2, "g/s/m2 × área da fonte", "g/s/m2", area_m2, length_m
  301          return np.nan, "área geométrica ausente - total não calculado", "g/s/m2", area_m2, length_m
  302      if typ in TOTAL_RATE_TYPES:
  303          return rate, "g/s total", "g/s", area_m2, length_m
  304      if typ in LENGTH_RATE_TYPES:
  305          if np.isfinite(length_m) and length_m > 0:
  306              return rate * length_m, "g/s/m × comprimento", "g/s/m", area_m2, length_m
  307          return np.nan, "comprimento ausente - total não calculado", "g/s/m", area_m2, length_m
  308      return rate, "taxa nativa - revisar", "desconhecida", area_m2, length_m

  368  # Associate enterprises
  369  def associate_enterprises(sources, enterprises, nearest_tolerance_m=250.0):
  370      out = sources.copy()
  371      out["enterprise"] = None
  372      if enterprises is None or enterprises.empty:
  373          return out
  374      emp = enterprises.to_crs(out.crs).copy()
  375      emp["geometry"] = emp.geometry.map(_clean_geometry)
  376      name_col = _enterprise_name_col(emp)
  377      if name_col is None:
  378          emp["_empname"] = [f"EMP_{i}" for i in emp.index]
  379      else:
  380          emp["_empname"] = emp[name_col].astype(str)
  381  
  382      for idx, src in out.iterrows():
  383          geom = src.geometry
  384          geom = _clean_geometry(geom)
  385          cand = emp[emp.intersects(geom)]
  386          chosen = None
  387          if not cand.empty:
  388              if geom.geom_type in {"Polygon", "MultiPolygon"}:
  389                  scores = cand.geometry.intersection(geom).area
  390                  chosen = cand.loc[scores.idxmax(), "_empname"]
  391              elif geom.geom_type in {"LineString", "MultiLineString"}:
  392                  scores = cand.geometry.intersection(geom).length
  393                  chosen = cand.loc[scores.idxmax(), "_empname"]
  394              else:
  395                  chosen = cand.iloc[0]["_empname"]
  396          else:
  397              d = emp.geometry.distance(geom)
  398              if len(d) and float(d.min()) <= nearest_tolerance_m:
  399                  chosen = emp.loc[d.idxmin(), "_empname"]
  400          out.at[idx, "enterprise"] = chosen
  401  
  402          subgroup = out.at[idx, "urban_subgroup"]
  403          has_urban_subgroup = pd.notna(subgroup) and str(subgroup).strip() != ""
  404          if has_urban_subgroup:
  405              out.at[idx, "group_id"] = "URB"
  406          elif chosen:
  407              g = canonical_group(chosen)
  408              if g and g not in {"AEX", "URB-FER", "URB-NUC", "URB-VIA"}:
  409                  out.at[idx, "group_id"] = g
  410      return out

  412  # Allocate emissions to zones
  413  def allocate_emissions_to_zones(sources, zones):
  414      """Aloca emissões às zonas preservando massa e as unidades nativas da fonte.
  415  
  416      Diferentemente da normalização pela soma das interseções, fontes parcialmente fora do
  417      zoneamento não têm sua emissão externa redistribuída artificialmente para as zonas internas.
  418      """
  419      z = zones.copy()
  420      z["EMIS_TOTAL_GS"] = 0.0
  421      z["AREA_KM2"] = z.geometry.area / 1e6
  422  
  423      allocations = []
  424      point_radius = float(getattr(config, "EMISSION_POINT_SUPPORT_RADIUS_M", 250.0))
  425      line_half_width = float(getattr(config, "EMISSION_LINE_SUPPORT_HALF_WIDTH_M", 50.0))
  426      for _, src in sources.iterrows():
  427          geom = src.geometry
  428          if geom is None or geom.is_empty:
  429              continue
  430          typ = str(src.get("source_type", "")).upper()
  431          total = pd.to_numeric(pd.Series([src.get("emission_total_gs")]), errors="coerce").iloc[0]
  432          rate_area = pd.to_numeric(pd.Series([src.get("surface_rate_gs_m2")]), errors="coerce").iloc[0]
  433          rate_line = pd.to_numeric(pd.Series([src.get("line_rate_gs_m")]), errors="coerce").iloc[0]
  434          area_total = pd.to_numeric(pd.Series([src.get("source_area_m2")]), errors="coerce").iloc[0]
  435          length_total = pd.to_numeric(pd.Series([src.get("source_length_m")]), errors="coerce").iloc[0]
  436  
  437          cand = z[z.intersects(geom)].copy()
  438          if cand.empty:
  439              continue
  440  
  441          if geom.geom_type in {"Point", "MultiPoint"}:
  442  
  443              n = len(cand)
  444              for zid in cand.index:
  445                  value = float(total) / n if np.isfinite(total) and n else 0.0
  446                  z.at[zid, "EMIS_TOTAL_GS"] += value
  447                  allocations.append({
  448                      "zone_index": zid, "source_id": src["source_id"], "source_type": typ,
  449                      "group_id": src.get("group_id"), "enterprise": src.get("enterprise"),
  450                      "emission_gs": value, "source_total_gs": total, "allocated_fraction": 1.0 / n if n else np.nan,
  451                      "intersection_measure": np.nan, "source_measure": np.nan, "allocation_basis": "emissão total g/s",
  452                      "source_intensity_gs_km2": value / (np.pi * point_radius**2 / 1e6) if point_radius > 0 else np.nan,
  453                  })
  454              continue
  455  
  456          if geom.geom_type in {"Polygon", "MultiPolygon"}:
  457              inter = cand.geometry.intersection(geom).area
  458              for zid, inter_area in inter.items():
  459                  if inter_area <= 0:
  460                      continue
  461                  if np.isfinite(rate_area):
  462                      value = float(rate_area) * float(inter_area)
  463                      frac = float(inter_area) / float(area_total) if np.isfinite(area_total) and area_total > 0 else np.nan
  464                      basis = "g/s/m2 × área de interseção"
  465                  elif np.isfinite(total) and np.isfinite(area_total) and area_total > 0:
  466                      frac = float(inter_area) / float(area_total)
  467                      value = float(total) * frac
  468                      basis = "g/s total × fração de área"
  469                  else:
  470                      continue
  471                  z.at[zid, "EMIS_TOTAL_GS"] += value
  472                  allocations.append({
  473                      "zone_index": zid, "source_id": src["source_id"], "source_type": typ,
  474                      "group_id": src.get("group_id"), "enterprise": src.get("enterprise"),
  475                      "emission_gs": value, "source_total_gs": total, "allocated_fraction": frac,
  476                      "intersection_measure": float(inter_area), "source_measure": area_total, "allocation_basis": basis,
  477                      "source_intensity_gs_km2": value / (float(inter_area) / 1e6) if inter_area > 0 else np.nan,
  478                  })
  479              continue
  480  
  481          if geom.geom_type in {"LineString", "MultiLineString"}:
  482              inter = cand.geometry.intersection(geom).length
  483              for zid, inter_len in inter.items():
  484                  if inter_len <= 0:
  485                      continue
  486                  if typ == "RLINEXT" and np.isfinite(src.get("emission_rate_native", np.nan)):
  487                      value = float(src["emission_rate_native"]) * float(inter_len)
  488                      frac = float(inter_len) / float(length_total) if np.isfinite(length_total) and length_total > 0 else np.nan
  489                      basis = "g/s/m × comprimento de interseção"
  490                  elif typ == "LINE_VOLUME" and np.isfinite(total) and np.isfinite(length_total) and length_total > 0:
  491                      frac = float(inter_len) / float(length_total)
  492                      value = float(total) * frac
  493                      basis = "g/s total × fração de comprimento"
  494                  elif np.isfinite(rate_line):
  495                      value = float(rate_line) * float(inter_len)
  496                      frac = float(inter_len) / float(length_total) if np.isfinite(length_total) and length_total > 0 else np.nan
  497                      basis = "g/s/m × comprimento de interseção"
  498                  else:
  499                      continue
  500                  z.at[zid, "EMIS_TOTAL_GS"] += value
  501                  allocations.append({
  502                      "zone_index": zid, "source_id": src["source_id"], "source_type": typ,
  503                      "group_id": src.get("group_id"), "enterprise": src.get("enterprise"),
  504                      "emission_gs": value, "source_total_gs": total, "allocated_fraction": frac,
  505                      "intersection_measure": float(inter_len), "source_measure": length_total, "allocation_basis": basis,
  506                      "source_intensity_gs_km2": value / (float(inter_len) * 2 * line_half_width / 1e6) if inter_len > 0 and line_half_width > 0 else np.nan,
  507                  })
  508  
  509      z["EMIS_DENS_GS_KM2"] = z["EMIS_TOTAL_GS"] / z["AREA_KM2"].replace(0, np.nan)
  510      dens = pd.to_numeric(z["EMIS_DENS_GS_KM2"], errors="coerce")
  511      pos = dens[dens > 0]
  512      q33 = float(pos.quantile(1/3)) if not pos.empty else np.nan
  513      q67 = float(pos.quantile(2/3)) if not pos.empty else np.nan
  514      def _dens_class(v):
  515          if pd.isna(v) or v <= 0:
  516              return "sem_emissao"
  517          if pd.notna(q33) and v <= q33:
  518              return "baixa"
  519          if pd.notna(q67) and v <= q67:
  520              return "moderada"
  521          return "alta"
  522      z["EMIS_DENS_CLASS"] = dens.map(_dens_class)
  523      z["EMIS_DENS_Q33"] = q33
  524      z["EMIS_DENS_Q67"] = q67
  525  
  526      a = pd.DataFrame(allocations)
  527      if not a.empty:
  528          focal = a.groupby("zone_index")["source_intensity_gs_km2"].quantile(0.95)
  529          z["EMIS_HOTSPOT_P95"] = focal.reindex(z.index).fillna(0.0)
  530  
  531          local = a[pd.to_numeric(a["emission_gs"], errors="coerce") > 0].groupby(
  532              ["zone_index", "group_id"], dropna=False
  533          )["emission_gs"].sum().reset_index()
  534          local["rank"] = local.groupby("zone_index")["emission_gs"].rank(method="first", ascending=False)
  535          dominant = local[local["rank"] == 1].set_index("zone_index")
  536          z["GRUPO_EMISSOR_LOCAL_ID"] = dominant["group_id"].reindex(z.index)
  537          z["GRUPO_EMISSOR_LOCAL"] = z["GRUPO_EMISSOR_LOCAL_ID"].map(
  538              lambda g: GROUPS.get(g, {}).get("name", g) if pd.notna(g) else None
  539          )
  540  
  541          src_totals = sources.assign(
  542              group_id=sources["group_id"].fillna("NAO_IDENTIFICADO")
  543          ).groupby("group_id")["emission_total_gs"].sum()
  544          alloc_totals = a.assign(
  545              group_id=a["group_id"].fillna("NAO_IDENTIFICADO")
  546          ).groupby("group_id")["emission_gs"].sum()
  547          group_outside = (src_totals - alloc_totals.reindex(src_totals.index).fillna(0.0)).clip(lower=0.0)
  548          z["EMIS_DENTRO_MUN_GS"] = z["EMIS_TOTAL_GS"]
  549          z["EMIS_FORA_MUN_GS"] = z["GRUPO_EMISSOR_LOCAL_ID"].map(group_outside).fillna(0.0)
  550          group_total = z["GRUPO_EMISSOR_LOCAL_ID"].map(src_totals)
  551          z["CONTR_EXTERNA_PCT"] = 100 * z["EMIS_FORA_MUN_GS"] / group_total.replace(0, np.nan)
  552      else:
  553          z["EMIS_HOTSPOT_P95"] = 0.0
  554          z["GRUPO_EMISSOR_LOCAL_ID"] = None
  555          z["GRUPO_EMISSOR_LOCAL"] = None
  556          z["EMIS_DENTRO_MUN_GS"] = z["EMIS_TOTAL_GS"]
  557          z["EMIS_FORA_MUN_GS"] = 0.0
  558          z["CONTR_EXTERNA_PCT"] = np.nan
  559  
  560      q = float(getattr(config, "EMISSION_PRESSURE_QUANTILE", 0.75))
  561      total_pos = pd.to_numeric(z["EMIS_TOTAL_GS"], errors="coerce")
  562      total_pos = total_pos[total_pos > 0]
  563      focal_pos = pd.to_numeric(z["EMIS_HOTSPOT_P95"], errors="coerce")
  564      focal_pos = focal_pos[focal_pos > 0]
  565      thr_total = float(total_pos.quantile(q)) if not total_pos.empty else np.nan
  566      thr_focal = float(focal_pos.quantile(q)) if not focal_pos.empty else np.nan
  567      z["EMIS_THR_TOTAL_GS"] = thr_total
  568      focal_tolerance = float(getattr(config, "EMISSION_FOCAL_THRESHOLD_RELATIVE_TOLERANCE", 0.01))
  569      effective_thr_focal = thr_focal * (1.0 + focal_tolerance) if pd.notna(thr_focal) else np.nan
  570      z["EMIS_THR_HOTSPOT_P95"] = effective_thr_focal
  571  
  572      z["EMIS_ALTA_TOTAL"] = pd.to_numeric(z["EMIS_TOTAL_GS"], errors="coerce").gt(thr_total) if pd.notna(thr_total) else False
  573      z["EMIS_ALTA_FOCAL"] = pd.to_numeric(z["EMIS_HOTSPOT_P95"], errors="coerce").gt(effective_thr_focal) if pd.notna(effective_thr_focal) else False
  574  
  575      if not a.empty:
  576          aa = a.copy()
  577          aa["emission_gs"] = pd.to_numeric(aa["emission_gs"], errors="coerce").fillna(0.0)
  578          aa["_urban_road"] = aa["group_id"].eq("URB")
  579          split = aa.groupby(["zone_index", "_urban_road"])["emission_gs"].sum().unstack(fill_value=0.0)
  580          z["EMIS_URBANA_GS"] = split.get(True, pd.Series(dtype=float)).reindex(z.index).fillna(0.0)
  581          z["EMIS_INDUSTRIAL_GS"] = split.get(False, pd.Series(dtype=float)).reindex(z.index).fillna(0.0)
  582      else:
  583          z["EMIS_URBANA_GS"] = 0.0
  584          z["EMIS_INDUSTRIAL_GS"] = 0.0
  585  
  586      q_ind = float(getattr(config, "EMISSION_INDUSTRIAL_HIGH_QUANTILE", 0.67))
  587      q_urb = float(getattr(config, "EMISSION_URBAN_HIGH_QUANTILE", 0.90))
  588      ind_pos = pd.to_numeric(z["EMIS_INDUSTRIAL_GS"], errors="coerce")
  589      ind_pos = ind_pos[ind_pos > 0]
  590      urban_zone = z.get("macrozona", pd.Series("", index=z.index)).astype(str).str.contains("Urbana", case=False, na=False)
  591      urb_pos = pd.to_numeric(z.loc[urban_zone, "EMIS_URBANA_GS"], errors="coerce")
  592      urb_pos = urb_pos[urb_pos > 0]
  593      thr_ind = float(ind_pos.quantile(q_ind)) if not ind_pos.empty else np.nan
  594      thr_urb = float(urb_pos.quantile(q_urb)) if not urb_pos.empty else np.nan
  595      z["EMIS_THR_INDUSTRIAL_GS"] = thr_ind
  596      z["EMIS_THR_URBANA_GS"] = thr_urb
  597      z["EMIS_ALTA_INDUSTRIAL"] = pd.to_numeric(z["EMIS_INDUSTRIAL_GS"], errors="coerce").gt(thr_ind) if pd.notna(thr_ind) else False
  598      z["EMIS_ALTA_URBANA"] = urban_zone & (pd.to_numeric(z["EMIS_URBANA_GS"], errors="coerce").gt(thr_urb) if pd.notna(thr_urb) else False)
  599      z["EMIS_PRESSAO_ALTA"] = z["EMIS_ALTA_INDUSTRIAL"] | z["EMIS_ALTA_URBANA"]
  600      z["EMIS_PRESSAO_TIPO"] = np.select(
  601          [z["EMIS_ALTA_INDUSTRIAL"] & z["EMIS_ALTA_URBANA"], z["EMIS_ALTA_INDUSTRIAL"], z["EMIS_ALTA_URBANA"]],
  602          ["industrial_e_urbana", "industrial_mineral", "ressuspensao_viaria_urbana"],
  603          default="baixa_em_ambas",
  604      )
  605      z["EMIS_CRITERIO"] = z["EMIS_PRESSAO_TIPO"]
  606      ind_rank = pd.to_numeric(z["EMIS_INDUSTRIAL_GS"], errors="coerce").rank(pct=True).fillna(0.0)
  607      urb_rank = pd.to_numeric(z["EMIS_URBANA_GS"], errors="coerce").where(urban_zone, 0.0).rank(pct=True).fillna(0.0)
  608      z["EMIS_PRESSAO_SCORE"] = pd.concat([ind_rank, urb_rank], axis=1).max(axis=1)
  609      return z, a

  611  # Emission mass balance audit
  612  def emission_mass_balance_audit(sources, allocations):
  613      """Auditoria de conservação de massa entre fontes e zoneamento."""
  614      s = sources[[c for c in ["source_id","source_type","group_id","enterprise","native_unit","emission_rate_native","source_area_m2","source_length_m","emission_total_gs","geometry_source"] if c in sources.columns]].copy()
  615      if allocations is None or allocations.empty:
  616          s["allocated_gs"] = 0.0
  617      else:
  618          a = allocations.groupby("source_id", as_index=False)["emission_gs"].sum().rename(columns={"emission_gs":"allocated_gs"})
  619          s = s.merge(a, on="source_id", how="left")
  620          s["allocated_gs"] = s["allocated_gs"].fillna(0.0)
  621      s["unallocated_gs"] = pd.to_numeric(s["emission_total_gs"], errors="coerce") - pd.to_numeric(s["allocated_gs"], errors="coerce")
  622      s["allocated_pct"] = 100 * pd.to_numeric(s["allocated_gs"], errors="coerce") / pd.to_numeric(s["emission_total_gs"], errors="coerce").replace(0, np.nan)
  623      return s

  625  # Group count audit
  626  def group_count_audit(sources):
  627      rows = []
  628      for group, meta in GROUPS.items():
  629          if group == "AEX":
  630              detected = int(sources["AEX"].fillna(False).sum()) if "AEX" in sources else 0
  631          elif group in {"URB-FER", "URB-NUC", "URB-VIA"}:
  632              detected = int((sources.get("urban_subgroup") == group).sum())
  633          else:
  634              detected = int((sources.get("group_id") == group).sum())
  635          expected = int(meta["expected"])
  636          rows.append({
  637              "group_id": group,
  638              "name": meta["name"],
  639              "kind": meta.get("kind"),
  640              "expected": expected,
  641              "detected": detected,
  642              "difference": detected - expected,
  643          })
  644      return pd.DataFrame(rows)
