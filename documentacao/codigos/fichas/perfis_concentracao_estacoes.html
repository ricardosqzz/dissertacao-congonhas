   90  # Dry interval
   91  def dry_interval(series,reference):
   92      """Maior queda cronológica da anomalia acumulada P - referência."""
   93      values=series.to_numpy(float)
   94      if not np.isfinite(values).all():raise ValueError('Não inferir estação seca preenchendo chuva ausente com zero')
   95      anomaly=np.r_[0,np.cumsum(values-reference)]
   96      end=int(np.argmax(np.maximum.accumulate(anomaly)-anomaly))
   97      if end==0:raise ValueError('Sem intervalo de déficit pluviométrico')
   98      start=int(np.argmax(anomaly[:end]))
   99      return start,end,float(anomaly[start]-anomaly[end])
