   44  def apply_inventory(sources,pol):
   45      inv=pd.read_excel(PATH,header=None)
   46      assert inv.iloc[0,22]=='Taxa de Emissão [ton/ano]'
   47      col={'PM10':23,'PM25':24}[pol]
   48      expected={'PM10':'MP10','PM25':'MP2,5'}[pol]
   49      assert inv.iloc[1,col]==expected
   50      inv=inv.iloc[2:].copy();inv['row']=inv.index+1
   51      inv['key']=inv[4].map(norm_token)
   52      inv['group']=inv[3].map(canonical_group)
   53      inv.loc[inv[1].eq('Emissão Urbana'),'group']='URB'
   54      inv['description']=inv[5].map(norm_token)
   55      inv['tpy']=pd.to_numeric(inv[col],errors='coerce')
   56      if inv.key.duplicated().any():raise ValueError('IDs duplicados no inventário')
   57      ids=inv.set_index('key');desc={}
   58      for key,sub in inv.groupby(['group','description']):
   59          if len(sub)==1:desc[key]=sub.iloc[0]
   60      out=sources.copy();out['inventory_row']=np.nan;out['inventory_source_id']=''
   61      out['inventory_match']='sem correspondência inequívoca'
   62      out['inventory_coordinate_distance_m']=np.nan
   63      out['model_emission_total_gs']=out.emission_total_gs
   64      for idx,r in out.iterrows():
   65          key=norm_token(r.source_id);match=None;basis=''
   66          if key in ids.index and ids.loc[key,'group']==r.group_id:
   67              match=ids.loc[key];basis='ID exato e grupo'
   68          elif (r.group_id,norm_token(r.desc)) in desc:
   69              match=desc[(r.group_id,norm_token(r.desc))];basis='descrição única no grupo'
   70          if match is not None and pd.notna(match.tpy) and match.tpy>=0:
   71              out.loc[idx,['inventory_row','inventory_source_id','inventory_match']]=[match.row,match[4],basis]
   72      out['emission_annual_tpy']=out.emission_total_gs*31.536
   73      out['emission_basis']='AERMOD anualizado 8760 h; conferir inventário'
   74      for row,sub in out.dropna(subset=['inventory_row']).groupby('inventory_row'):
   75          total=float(inv.loc[inv.row.eq(row),'tpy'].iloc[0])
   76  
   77          weights=sub.model_emission_total_gs.clip(lower=0)
   78          if len(sub)>1 and (not np.isfinite(weights).all() or weights.sum()<=0):
   79  
   80              if sub.geometry.geom_type.isin(['Polygon','MultiPolygon']).all():
   81                  weights=sub.geometry.area
   82              elif sub.geometry.geom_type.isin(['LineString','MultiLineString']).all():
   83                  weights=sub.geometry.length
   84              else:
   85                  out.loc[sub.index,'inventory_match']='partição sem pesos válidos; mantida taxa AERMOD'
   86                  continue
   87              if not np.isfinite(weights).all() or weights.sum()<=0:
   88                  out.loc[sub.index,'inventory_match']='partição sem medida geométrica válida'
   89                  continue
   90              out.loc[sub.index,'inventory_match']=out.loc[sub.index,'inventory_match']+'; partição por medida geométrica'
   91          weights=weights/weights.sum() if len(sub)>1 else pd.Series(1.,index=sub.index)
   92  
   93          from shapely.geometry import Point
   94          from shapely.ops import unary_union
   95          inventory_record=inv.loc[inv.row.eq(row)].iloc[0]
   96          x=pd.to_numeric(inventory_record[6],errors='coerce')
   97          y=pd.to_numeric(inventory_record[7],errors='coerce')
   98          if np.isfinite(x) and np.isfinite(y):
   99              distance=unary_union(sub.geometry.tolist()).distance(Point(x,y))
  100              out.loc[sub.index,'inventory_coordinate_distance_m']=distance
  101              if distance>5000:
  102                  out.loc[sub.index,'inventory_match']='divergência espacial >5 km; mantida taxa AERMOD'
  103                  continue
  104          out.loc[sub.index,'emission_annual_tpy']=total*weights
  105          out.loc[sub.index,'emission_basis']='Inventário 2023 t/ano'
  106  
  107      out['emission_total_gs']=out.emission_annual_tpy/31.536
  108      polygon=out.geometry.geom_type.isin(['Polygon','MultiPolygon'])
  109      line=out.geometry.geom_type.isin(['LineString','MultiLineString'])
  110      out.loc[polygon,'surface_rate_gs_m2']=out.loc[polygon,'emission_total_gs']/out.loc[polygon,'source_area_m2']
  111      out.loc[line,'line_rate_gs_m']=out.loc[line,'emission_total_gs']/out.loc[line,'source_length_m']
  112  
  113      sel=line&out.source_type.eq('RLINEXT')
  114      out.loc[sel,'emission_rate_native']=out.loc[sel,'line_rate_gs_m']
  115      audit=out.drop(columns='geometry').copy()
  116      audit['inventory_file']=str(PATH);audit['inventory_year']=2023
  117      inv['matched_geometry']=inv.row.isin(out.loc[out.emission_basis.eq('Inventário 2023 t/ano'),'inventory_row'])
  118      missing=inv.loc[inv.tpy.gt(0)&~inv.matched_geometry,[0,3,4,5,'row','tpy','group']].copy()
  119      missing.columns=['municipality','enterprise','inventory_id','description','inventory_row','tpy','group_id']
  120      return out,audit,missing

  123  def calculate_zone_emissions(sources,zones,pol,output):
  124      """Inventário anual conciliado; resíduos ficam explícitos, sem alterar POSTFILES."""
  125      from aermod_sources import allocate_emissions_to_zones,emission_mass_balance_audit
  126      from methodology_rules import group_mass_classes
  127      from pathlib import Path
  128      output=Path(output);output.mkdir(parents=True,exist_ok=True)
  129      sources,audit,missing=apply_inventory(sources,pol)
  130      _,alloc=allocate_emissions_to_zones(sources,zones)
  131      meta=sources.set_index('source_id')
  132      for col in ('inventory_source_id','inventory_row','inventory_match','emission_basis'):
  133          alloc[col]=alloc.source_id.map(meta[col])
  134      alloc['emission_tpy']=alloc.allocated_fraction*alloc.source_id.map(meta.emission_annual_tpy)
  135      mass,groups=group_mass_classes(alloc,zones.index,pol,annualization_basis='inventario_conciliado')
  136      mass['emission_inventory_tpy']=alloc.loc[alloc.emission_basis.eq('Inventário 2023 t/ano')].groupby('zone_index').emission_tpy.sum().reindex(zones.index,fill_value=0)
  137      mass['emission_fallback_tpy']=mass.emission_total_tpy-mass.emission_inventory_tpy
  138      mass['emission_inventory_pending_sources']=len(missing)
  139      mass['emission_audit_status']='Inventário conciliado parcialmente; consultar fontes pendentes'
  140      balance=emission_mass_balance_audit(sources,alloc)
  141      balance['source_total_tpy']=balance.emission_total_gs*31.536
  142      balance['allocated_tpy']=balance.allocated_gs*31.536
  143      balance['unallocated_tpy']=balance.unallocated_gs*31.536
  144      balance['overlap_warning']=balance.allocated_pct.gt(100.000001)
  145      balance['rate_basis']='g/s equivalente da massa anual; não substitui entrada AERMOD'
  146      for name,table in [('alocacao_fontes',alloc),('massa_grupo_zona',groups),('balanco_massa',balance),('conciliacao_inventario',audit),('inventario_sem_geometria_conciliada',missing)]:
  147          table.to_csv(output/f'{pol}_{name}.csv',index=False,encoding='utf-8-sig')
  148      return mass,groups,alloc,audit,missing
