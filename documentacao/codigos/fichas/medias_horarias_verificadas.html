   58  # Model daily mean
   59  def model_daily_mean(hourly_sum,valid_hours):
   60   """Política de calmarias para 24 h: denominador válido, mínimo de 18."""
   61   if np.any(np.asarray(valid_hours)<0) or np.any(np.asarray(valid_hours)>24):raise ValueError('Contagem diária fora de 0–24 h')
   62   return np.asarray(hourly_sum)/np.maximum(valid_hours,18)

   69  def met_calendar():

   72   code=((f[0]*100+f[1])*100+f[2])*100+f[4]
   73   valid=f[15].gt(0)&f[6].gt(0)&f[18].gt(0)&f[11].ne(0)&f[11].gt(-99990)
   74  
   75   valid &= np.where(f[5].gt(0),f[7].gt(0)&f[9].gt(0),f[10].gt(0))
   76   profile=pd.read_csv(p,sep=r'\s+',header=None)
   77   pc=((profile[0]*100+profile[1])*100+profile[2])*100+profile[3]
   78   profile_ok=profile[7].ge(0)&profile[7].lt(90)&profile[8].gt(-90)&profile[8].lt(90)
   79   profile_ok=pd.Series(profile_ok.to_numpy(),index=pc).groupby(level=0).any()
