  110  def concentration_priority(statuses):
  111      """P0–P5 da menor à maior superação; lacuna impede classe conclusiva."""
  112      values=[statuses.get(stage) for stage in STAGE_ORDER]
  113      if any(v not in ('atendimento','nao_atendimento') for v in values):
  114          return 'Informação insuficiente'
  115      return 'P'+str(max([0]+[i for i,v in enumerate(values,1) if v=='nao_atendimento']))

  126  def model_annual_metrics(daily):
  127      """Indicadores pareados por receptor/ano; 75% em CADA trimestre."""
  128      from standards import annual_p98
  129      d=daily[['X','Y','date','CONC']].copy()
  130      d['date']=pd.to_datetime(d.date).dt.normalize()
  131      if d.duplicated(['X','Y','date']).any():raise ValueError('Receptor/data duplicado entre POSTFILEs.')
  132      d['CONC']=pd.to_numeric(d.CONC,errors='coerce')
  133      d.loc[~np.isfinite(d.CONC)|(d.CONC<0),'CONC']=np.nan
  134      d['year']=d.date.dt.year;d['quarter']=d.date.dt.quarter
  135      result=d.groupby(['X','Y','year']).agg(ANNUAL_MEAN=('CONC','mean'),P98=('CONC',annual_p98),N_DAYS=('CONC','count'))
  136      counts=d.groupby(['X','Y','year','quarter']).CONC.count().unstack('quarter').reindex(columns=[1,2,3,4],fill_value=0).fillna(0)
  137      for quarter in range(1,5):
  138          expected=[(pd.Timestamp(int(y),quarter*3,1)+pd.offsets.MonthEnd(0)-pd.Timestamp(int(y),quarter*3-2,1)).days+1 for y in counts.index.get_level_values('year')]
  139          counts[quarter]=counts[quarter]/np.asarray(expected)
  140      result['MIN_QUARTER_COMPLETENESS']=counts.min(axis=1)
  141      result['VALID_YEAR']=result.MIN_QUARTER_COMPLETENESS.ge(.75)

  161  # Abc classes
  162  def abc_classes(values, a_limit=80.0, b_limit=95.0):
  163      """Inclui o grupo que cruza o limite; empates exatos ficam na mesma classe.
  164  
  165      A classificação usa o acumulado ANTES do bloco de valores empatados.
  166      A entrada deve estar em ordem decrescente, como nas tabelas ABC existentes.
  167      Zero fica em C; ausência, negativo e infinito não são contribuições válidas.
  168      """
  169      if not 0 < a_limit < b_limit < 100:
  170          raise ValueError('Limites ABC devem satisfazer 0 < A < B < 100.')
  171      v=np.asarray(values,dtype=float)
  172      if np.any(~np.isfinite(v)) or np.any(v<0):
  173          raise ValueError('Contribuições devem ser finitas e não negativas.')
  174      if np.any(np.diff(v)>0):
  175          raise ValueError('Ordene as contribuições de forma decrescente.')
  176      if not len(v) or v.sum()<=0:
  177          return np.full(len(v),'insuficiente',dtype=object)
  178      result=[];before=0.0
  179      for i,x in enumerate(v):
  180          if i==0 or x!=v[i-1]:
  181              cls='C' if x==0 else ('A' if before<a_limit-1e-10 else 'B' if before<b_limit-1e-10 else 'C')
  182          result.append(cls)
  183          before+=100.0*x/v.sum()
  184      return np.asarray(result,dtype=object)

  186  # Rolling station metrics
  187  def rolling_station_metrics(yearly, start_year=2018, end_year=None):
  188      """Uma linha por estação/poluente/janela; nunca pula anos inválidos.
  189  
  190      Consome a tabela anual validada de stations.station_yearly_metrics.
  191      Não transforma ausência de dados em atendimento e não cria anos futuros.
  192      """
  193      if yearly is None or yearly.empty:return pd.DataFrame()
  194      keys=['station_id','pollutant','year']
  195      if yearly.duplicated(keys).any():raise ValueError('Estação/poluente/ano duplicado.')
  196      observed_end=int(yearly.year.max())
  197      last=observed_end if end_year is None else int(end_year)
  198      rows=[]
  199      for (sid,pol),group in yearly.groupby(keys[:2],sort=True):
  200          if pol not in STANDARDS:continue
  201          byyear=group.set_index('year')
  202          for end in range(start_year+2,last+1):
  203              years=list(range(end-2,end+1));d=byyear.reindex(years)
  204              valid=d.valid_year.eq(True) & d.annual_mean.notna() & d.p98.notna()
  205              valid &= np.isfinite(d.annual_mean) & np.isfinite(d.p98)
  206              complete=bool(valid.all())
  207              ann=float(d.annual_mean.mean()) if complete else np.nan
  208              p98=float(d.p98.mean()) if complete else np.nan
  209              r={'station_id':sid,'pollutant':pol,'start_year':end-2,'end_year':end,
  210                 'period':f'{end-2}-{end}','n_design_years':int(valid.sum()),
  211                 'years_valid':','.join(str(y) for y in years) if complete else '',
  212                 'ANNUAL_3YR':ann,'P98_3YR':p98}
  213              for scenario in ('PV',)+STAGE_ORDER:
  214                  lim=STANDARDS[pol][scenario]
  215                  r['status_'+scenario]='insuficiente' if not complete else 'nao_atendimento' if ann>lim['ANNUAL'] or p98>lim['24H'] else 'atendimento'
  216              rows.append(r)
  217      return pd.DataFrame(rows)
