  101  def write_shp(gdf,path):
  102      import shapefile
  103      from io_utils import _shp_safe_columns,_sanitize_vector_columns
  104      gdf,aliases=_shp_safe_columns(_sanitize_vector_columns(gdf));cols=[c for c in gdf if c!=gdf.geometry.name]
  105      with shapefile.Writer(str(path),shapeType=shapefile.POLYGON,encoding='utf-8') as w:
  106          for col in cols:
  107              s=gdf[col]
  108              if pd.api.types.is_bool_dtype(s):w.field(col,'L')
  109              elif pd.api.types.is_numeric_dtype(s):w.field(col,'N',size=24,decimal=8)
  110              else:w.field(col,'C',size=254)
  111          from shapely.geometry import MultiPolygon
  112          for _,r in gdf.iterrows():
  113              geom=r.geometry
  114              if geom.geom_type=='MultiPolygon':geom=MultiPolygon([orient(p,sign=-1) for p in geom.geoms])
  115              elif geom.geom_type=='Polygon':geom=orient(geom,sign=-1)
  116              else:raise ValueError(f'Shapefile requer polígonos: {geom.geom_type}')
  117              w.shape(mapping(geom));w.record(*[None if pd.isna(r[c]) else r[c].item() if isinstance(r[c],np.generic) else r[c] for c in cols])
  118      Path(path).with_suffix('.prj').write_text(gdf.crs.to_wkt(version='WKT1_ESRI'))
  119      Path(path).with_suffix('.cpg').write_text('UTF-8')
  120      pd.DataFrame([{'campo_original':a,'campo_shp':b} for a,b in aliases.items()]).to_csv(Path(path).with_name(Path(path).stem+'_campos.csv'),index=False,encoding='utf-8-sig')
