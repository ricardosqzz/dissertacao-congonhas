   71  # Norm token
   72  def norm_token(value):
   73      text = unicodedata.normalize("NFKD", str(value or ""))
   74      text = "".join(ch for ch in text if not unicodedata.combining(ch))
   75      return re.sub(r"[^A-Z0-9]+", "", text.upper())
