   64  def rounded_scale_ticks(norm):
   65      """Marcas legíveis nos valores reais, sem arredondar apenas os rótulos."""
   66      unit=10.**np.floor(np.log10(norm.vmax))
   67      norm.vmax=np.ceil(norm.vmax/unit)*unit
   68      raw=norm.inverse(np.linspace(0,1,4))
   69      ticks=[0.]
   70      for value in raw[1:-1]:
   71          step=10.**np.floor(np.log10(value))
   72          ticks.append(float(np.round(value/step)*step))
   73      return np.unique(ticks+[float(norm.vmax)])
